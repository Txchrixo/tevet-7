   252→function Navbar({ t, onLogin, onSignup }: { t: typeof T.en; onLogin: () => void; onSignup: () => void }) {
   253→  const [scrolled, setScrolled] = useState(false);
   254→  const [mobileOpen, setMobileOpen] = useState(false);
   255→
   256→  useEffect(() => {
   257→    const onScroll = () => setScrolled(window.scrollY > 20);
   258→    window.addEventListener("scroll", onScroll);
   259→    return () => window.removeEventListener("scroll", onScroll);
   260→  }, []);
   261→
   262→  const navLinks = [
   263→    { label: t.navFeatures, href: "#features" },
   264→    { label: t.navHow, href: "#how" },
   265→    { label: t.navDemo, href: "#demo" },
   266→    { label: t.navPricing, href: "#pricing" },
   267→    { label: t.navFaq, href: "#faq" },
   268→  ];
   269→
   270→  return (
   271→    <nav className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${scrolled ? "bg-background/90 backdrop-blur-md border-b border-border" : "bg-transparent"}`}>
   272→      <div className="max-w-6xl mx-auto px-4 h-16 flex items-center justify-between">
   273→        {/* Logo: clickable → scroll to top (home) */}
   274→        <a href="#" onClick={(e) => { e.preventDefault(); window.scrollTo({ top: 0, behavior: "smooth" }); }} className="cursor-pointer">
   275→          <BrandLogo size={26} />
   276→        </a>
   277→        {/* Nav links: hidden on < md */}
   278→        <div className="hidden md:flex items-center gap-6">
   279→          {navLinks.map((l) => (<a key={l.href} href={l.href} className="text-sm text-muted-foreground hover:text-foreground transition-colors">{l.label}</a>))}
   280→        </div>
   281→        {/* CTAs: "Log in" hidden on < sm, "Try free" always visible.
   282→            "Log in" opens the AuthScreen in login mode; "Try free" opens
   283→            it in signup mode. Neither starts the demo - the demo is only
   284→            triggered from the Hero's "View demo" button. */}
   285→        <div className="flex items-center gap-2 sm:gap-3">
   286→          <button onClick={onLogin} className="hidden sm:block text-sm text-muted-foreground hover:text-foreground transition-colors">{t.navLogin}</button>
   287→          <button onClick={onSignup} className="rounded-md bg-primary px-3 sm:px-4 py-1.5 text-sm font-medium text-foreground transition-colors hover:bg-accent hover:text-accent-foreground whitespace-nowrap">{t.navTry}</button>
   288→          {/* Hamburger: only on < md */}
   289→          <button className="md:hidden text-foreground ml-1" onClick={() => setMobileOpen(!mobileOpen)}>
   290→            {mobileOpen ? <X className="h-5 w-5" /> : <Menu className="h-5 w-5" />}
   291→          </button>
   292→        </div>
   293→      </div>
   294→      <AnimatePresence>
   295→        {mobileOpen && (
   296→          <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} exit={{ opacity: 0, height: 0 }} className="md:hidden bg-background border-b border-border overflow-hidden">
   297→            <div className="px-4 py-4 space-y-3">
   298→              {navLinks.map((l) => (<a key={l.href} href={l.href} onClick={() => setMobileOpen(false)} className="block text-sm text-muted-foreground hover:text-foreground">{l.label}</a>))}
   299→              <button onClick={() => { setMobileOpen(false); onLogin(); }} className="block w-full text-left text-sm text-muted-foreground hover:text-foreground">{t.navLogin}</button>
   300→            </div>
   301→          </motion.div>
   302→        )}
   303→      </AnimatePresence>
   304→    </nav>
   305→  );
   306→}

   389→function renderSqlTokens(text: string) {
   390→  const tokenRe = /('(?:[^']|'')*')|(\b\d+(?:\.\d+)?\b)|([A-Za-z_][A-Za-z0-9_]*)|([(),.;*=<>+\-/]+)|(\s+)|([^\s])/g;
   391→  const nodes: React.ReactNode[] = [];
   392→  let m: RegExpExecArray | null; let key = 0;
   393→  while ((m = tokenRe.exec(text)) !== null) {
   394→    const [full, str, num, ident, punct, ws] = m;
   395→    if (str) nodes.push(<span key={key++} className="text-foreground/90">{str}</span>);
   396→    else if (num) nodes.push(<span key={key++} className="font-heading text-muted-foreground tabular-nums">{num}</span>);
   397→    else if (ident) { const upper = ident.toUpperCase(); if (SQL_KEYWORDS.has(upper)) nodes.push(<span key={key++} className="font-medium text-accent/90">{ident}</span>); else nodes.push(<span key={key++} className="text-foreground">{ident}</span>); }
   398→    else if (punct) nodes.push(<span key={key++} className="text-muted-foreground">{punct}</span>);
   399→    else if (ws) nodes.push(ws);
   400→    else nodes.push(full);
   401→  }
   402→  return nodes;
   403→}
