   183→/** Returns "MD" for "Marie Dubois" etc. — used to derive initials from a real user name. */
   184→function initialsFromName(name: string): string {
   185→  const parts = name.trim().split(/\s+/).filter(Boolean);
   186→  if (parts.length === 0) return "?";
   187→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   188→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   189→}

   208→    producerId,

   211→    initials: initialsFromName(user.name),
   212→    accent: "accent",

   216→/**
   217→ * Adapts the backend `/api/chat` envelope (snake_case) to the frontend's
   218→ * `AssistantResponse` (camelCase). Unknown / missing fields fall back to safe
   219→ * defaults so the inspector never crashes.
   220→ *
   221→ * Field conversions:
   222→ *   - `scope_clause`  → `scopeClause`
   223→ *   - `tokens_in`     → `tokensIn`
   224→ *   - `tokens_out`    → `tokensOut`
   225→ *   - `latency_ms`    → `latencyMs`
   226→ *   - `tool_calls`    → `toolCalls`
   227→ *   - `security_checks` → `securityChecks` (fields are already camelCase)
   228→ *   - `steps[].duration_ms` → `steps[].durationMs`
   229→ */
   230→function adaptBackendResponse(raw: unknown): AssistantResponse {
   231→  const r =
   232→    (typeof raw === "object" && raw !== null ? raw : {}) as Record<
   233→      string,
   234→      unknown
   235→    >;
   236→  const rawSteps = Array.isArray(r.steps) ? (r.steps as Record<string, unknown>[]) : [];
   237→  const steps: TraceStep[] = rawSteps.map((s) => ({
   238→    index: typeof s.index === "number" ? s.index : 0,
   239→    title: typeof s.title === "string" ? s.title : "",
   240→    detail: typeof s.detail === "string" ? s.detail : "",
   241→    status: (s.status === "ok" || s.status === "warning" || s.status === "blocked"
   242→      ? s.status
   243→      : "ok") as TraceStep["status"],
   244→    durationMs: typeof s.duration_ms === "number" ? s.duration_ms : 0,
   245→  }));
   246→  const rawChecks = Array.isArray(r.security_checks)
   247→    ? (r.security_checks as Record<string, unknown>[])
   248→    : [];
   249→  const securityChecks: AssistantResponse["securityChecks"] = rawChecks.map((c) => ({
   250→    label: typeof c.label === "string" ? c.label : "",
   251→    status: (c.status === "ok" || c.status === "warning" || c.status === "blocked"
   252→      ? c.status
   253→      : "ok") as AssistantResponse["securityChecks"][number]["status"],
   254→    detail: typeof c.detail === "string" ? c.detail : undefined,
   255→  }));
   256→  const toolCalls = Array.isArray(r.tool_calls)
   257→    ? (r.tool_calls as string[])
   258→    : [];
   259→  return {
   260→    answer: typeof r.answer === "string" ? r.answer : "Réponse vide.",
   261→    sql: typeof r.sql === "string" ? r.sql : null,
   262→    scopeClause:
   263→      typeof r.scope_clause === "string" ? r.scope_clause : null,
   264→    chart:
   265→      (r.chart as AssistantResponse["chart"] | null | undefined) ?? undefined,
   266→    tokensIn: typeof r.tokens_in === "number" ? r.tokens_in : 0,
   267→    tokensOut: typeof r.tokens_out === "number" ? r.tokens_out : 0,
   268→    latencyMs: typeof r.latency_ms === "number" ? r.latency_ms : 0,
   269→    toolCalls,
   270→    steps,
   271→    securityChecks,
   272→    refused: Boolean(r.refused),
   273→  };
   274→}

   276→/** Calls `POST /api/chat` with the message + JWT, returns an `AssistantResponse`.
   277→ *
   278→ * Demo override: when `demoIdentityOverride` is set (Marie/Pierre/Admin switcher
   279→ * on the demo tenant), sends body identity instead of JWT so the user can switch
   280→ * without re-login. The backend's dual-mode fallback handles the scoping.
   281→ */
   282→async function callBackendChat(message: string): Promise<AssistantResponse> {
   283→  const token = getAuthToken();
   284→  const override = useCopilotStore.getState().demoIdentityOverride;
   285→  const headers: Record<string, string> = {
   286→    "content-type": "application/json",
   287→  };
   288→  // When a demo identity override is active, send body identity (no JWT) so
   289→  // the backend uses the override's producer_id/role instead of the JWT's.
   290→  const useJwt = token && !override;
   291→  if (useJwt) headers["authorization"] = `Bearer ${token}`;
   292→
   293→  const body: Record<string, unknown> = { message };
   294→  if (override) {
   295→    body.identity_id = override.id;
   296→    body.producer_id = override.producerId;
   297→    body.role = override.kind;
   298→  }
   299→
   300→  const res = await fetch("/api/chat", {
   301→    method: "POST",
   302→    headers,
   303→    body: JSON.stringify(body),
   304→  });
   305→  const text = await res.text();
   306→  let parsed: unknown = null;
   307→  if (text.length > 0) {
   308→    try {
   309→      parsed = JSON.parse(text);
   310→    } catch {
   311→      parsed = text;
   312→    }
   313→  }
   314→  if (!res.ok) {
   315→    const message =
   316→      typeof parsed === "object" && parsed && "detail" in parsed
   317→        ? String((parsed as { detail: unknown }).detail)
   318→        : typeof parsed === "object" && parsed && "message" in parsed
   319→          ? String((parsed as { message: unknown }).message)
   320→          : `Chat backend ${res.status} ${res.statusText}`;
   321→    throw new Error(message);
   322→  }
   323→  return adaptBackendResponse(parsed);
   324→}

   351→  /**
   352→   * Called on app start. If we have a stored JWT, validate it via `/api/auth/me`
   353→   * and load the user's tenants. If the JWT is invalid or absent, switch to
   354→   * `anonymous` so the AuthScreen renders. Network failures fall back to
   355→   * `anonymous` (the user can still click "Essayer la démo").
   356→   */
   357→  bootstrap: async () => {
   358→    const token = getAuthToken();
   359→    if (!token) {
   360→      set({ authMode: "anonymous" });
   361→      return;
   362→    }
   363→    try {
   364→      const me = await getMe();
   365→      const tenants = me.memberships ?? [];
   366→      const activeTenant = pickActiveTenant(tenants);
   367→      set({
   368→        authMode: "authenticated",
   369→        isDemoSession: false,
   370→        user: me.user,
   371→        tenants,
   372→        activeTenant,
   373→        identity: identityFromAuthUser(me.user, activeTenant),
   374→      });
   375→    } catch (err) {
   376→      // Stale or invalid JWT — clear it and show the auth screen.
   377→      setAuthToken(null);
   378→      setActiveTenantId(null);
   379→      set({ authMode: "anonymous", user: null, tenants: [], activeTenant: null });
   380→      if (err instanceof AuthApiError && !err.unreachable) {
   381→        // 401 / 403 etc. — silently clear. Other errors (502) are also silent
   382→        // so the user just sees the auth screen.
   383→      }
   384→    }
   385→  },

   387→  /**
   388→   * Real backend login. Stores the JWT, loads `/api/auth/me` + `/api/tenants/mine`,

   391→   * Returns `{ ok: false, error }` on failure so the AuthScreen can show an
   392→   * inline error message. Network failures (backend unreachable) are surfaced
   393→   * as `error` so the caller can decide whether to fall back to demo mode.

   406→        authMode: "authenticated",
   407→        isDemoSession: false,
   408→        authLoading: false,

   410→        tenants,

   554→  /**
   555→   * Switches the active tenant. Calls `POST /api/tenants/{id}/activate` to get
   556→   * a fresh JWT scoped to the new tenant, then re-derives the chat identity.
   557→   */
   558→  switchTenant: async (tenantId) => {
   559→    const token = getAuthToken();
   560→    if (!token) return;
   561→    try {
   562→      const { membership, token: newToken } = await activateTenant(tenantId);
   563→      setAuthToken(newToken);
   564→      setActiveTenantId(membership.tenant_id);
   565→      const tenants = get().tenants.map((t) =>
   566→        t.tenant_id === membership.tenant_id
   567→          ? { ...t, is_active: true }
   568→          : { ...t, is_active: false },
   569→      );
   570→      const activeTenant = tenants.find((t) => t.tenant_id === membership.tenant_id) ?? null;
   571→      const user = get().user;
   572→      set({
   573→        tenants,
   574→        activeTenant,
   575→        identity: user ? identityFromAuthUser(user, activeTenant) : get().identity,
   576→        messages: [],
   577→        selectedMessageId: null,
   578→        isStreaming: false,
   579→        inspectorOpen: false,
   580→        adminView: "none",
   581→      });
   582→      toast.success(`Tenant activé : ${activeTenant?.name ?? tenantId}`);
   583→    } catch (err) {
   584→      const message =
   585→        err instanceof AuthApiError
   586→          ? err.unreachable
   587→            ? "Backend injoignable"
   588→            : err.message
   589→          : err instanceof Error
   590→            ? err.message
   591→            : "Erreur inconnue";
   592→      toast.error("Impossible de changer de tenant", { description: message });
   593→    }
   594→  },

   809→/**
   810→ * Picks the tenant to activate after login: the cached `activeTenantId` if the
   811→ * user is still a member, else the first `is_active` membership, else the
   812→ * first membership. Returns null if the user has no memberships.
   813→ */
   814→function pickActiveTenant(tenants: TenantMembership[]): TenantMembership | null {
   815→  if (tenants.length === 0) return null;
   816→  const cachedId = getActiveTenantId();
   817→  if (cachedId) {
   818→    const cached = tenants.find((t) => t.tenant_id === cachedId);
   819→    if (cached) return cached;
   820→  }
   821→  const active = tenants.find((t) => t.is_active);
   822→  if (active) return active;
   823→  return tenants[0];
   824→}

   826→/**
   827→ * Shared helper that pushes the user message, simulates the "thinking" delay,
   828→ * then reveals the assistant response. Responses are computed up-front from
   829→ * the mock layer so they stay deterministic per identity + question.
   830→ */
   831→function runAssistant(
   832→  get: () => CopilotState,
   833→  set: (partial: Partial<CopilotState>) => void,
   834→  userText: string,
   835→  computeResponse: () => AssistantResponse,
   836→) {
   837→  if (get().isStreaming) return;
   838→
   839→  const userMessage: ChatMessage = {
   840→    id: makeId("u"),
   841→    role: "user",
   842→    content: userText,
   843→    createdAt: Date.now(),
   844→  };
   845→
   846→  set({
   847→    messages: [...get().messages, userMessage],
   848→    isStreaming: true,
   849→    selectedMessageId: null,
   850→  });
   851→
   852→  // Simulate model latency before revealing the answer.
   853→  const delay = 800 + Math.random() * 350;
   854→  window.setTimeout(() => {
   855→    const response = computeResponse();
   856→    const assistantMessage: ChatMessage = {
   857→      id: makeId("a"),
   858→      role: "assistant",
   859→      content: response.answer,
   860→      response,
   861→      createdAt: Date.now(),
   862→      streaming: false,
   863→    };
   864→    // If the inspector is already open, live-update it with the new trace.
   865→    // Otherwise leave it closed — the user can click the answer to inspect.
   866→    const inspectorWasOpen = get().inspectorOpen;
   867→    set({
   868→      messages: [...get().messages, assistantMessage],
   869→      isStreaming: false,
   870→      selectedMessageId: inspectorWasOpen ? assistantMessage.id : null,
   871→    });
   872→  }, delay);
   873→}

   875→/**
   876→ * Backend-backed assistant run. Pushes the user message immediately, shows the
   877→ * typing indicator, calls `/api/chat` with the JWT, then pushes the assistant
   878→ * message with the real envelope (SQL, chart, steps, security_checks).
   879→ *
   880→ * Network / HTTP errors are surfaced as a synthetic assistant message so the
   881→ * chat thread doesn't break.
   882→ */
   883→async function runBackendAssistant(
   884→  get: () => CopilotState,
   885→  set: (partial: Partial<CopilotState>) => void,
   886→  userText: string,
   887→) {
   888→  if (get().isStreaming) return;
   889→
   890→  const userMessage: ChatMessage = {
   891→    id: makeId("u"),
   892→    role: "user",
   893→    content: userText,
   894→    createdAt: Date.now(),
   895→  };
   896→
   897→  set({
   898→    messages: [...get().messages, userMessage],
   899→    isStreaming: true,
   900→    selectedMessageId: null,
   901→  });
   902→
   903→  try {
   904→    const response = await callBackendChat(userText);
   905→    const assistantMessage: ChatMessage = {
   906→      id: makeId("a"),
   907→      role: "assistant",
   908→      content: response.answer,
   909→      response,
   910→      createdAt: Date.now(),
   911→      streaming: false,
   912→    };
   913→    const inspectorWasOpen = get().inspectorOpen;
   914→    set({
   915→      messages: [...get().messages, assistantMessage],
   916→      isStreaming: false,
   917→      selectedMessageId: inspectorWasOpen ? assistantMessage.id : null,
   918→    });
   919→  } catch (err) {
   920→    const message =
   921→      err instanceof Error ? err.message : "Erreur inconnue";
   922→    const failureResponse: AssistantResponse = {
   923→      answer:
   924→        "⚠️ Le service agent n'a pas pu répondre.\n\n" +
   925→        `Détail : ${message}\n\n` +
   926→        "Réessayez dans un instant. Si le problème persiste, basculez en mode démo via le menu utilisateur.",
   927→      sql: null,
   928→      scopeClause: null,
   929→      tokensIn: 0,
   930→      tokensOut: 0,
   931→      latencyMs: 0,
   932→      toolCalls: [],
   933→      steps: [],
   934→      securityChecks: [],
   935→      refused: true,
   936→    };
   937→    const assistantMessage: ChatMessage = {
   938→      id: makeId("a"),
   939→      role: "assistant",
   940→      content: failureResponse.answer,
   941→      response: failureResponse,
   942→      createdAt: Date.now(),
   943→      streaming: false,
   944→    };
   945→    set({
   946→      messages: [...get().messages, assistantMessage],
   947→      isStreaming: false,
   948→    });
   949→    toast.error("Échec de l'agent", { description: message });
   950→  }
   951→}
