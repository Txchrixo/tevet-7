   865→   865→ * Try to match a free-form user message to one of the example question ids,
   866→   866→ * otherwise fall back to a synthesised scoped response.
   867→   867→ */
   868→   868→export function getResponseForMessage(
   869→   869→  identity: Identity,
   870→   870→  message: string,
   871→   871→): AssistantResponse {
   872→   872→  const normalized = message.trim().toLowerCase();
   873→   873→  const match = EXAMPLE_QUESTIONS.find((q) =>
   874→   874→    normalized.includes(q.label.toLowerCase().slice(0, 25)),
   875→   875→  );
   876→   876→  if (match) {
   877→   877→    const key = `${identity.id}:${match.id}`;
   878→   878→    const r = RESPONSES[key];
   879→   879→    if (r) return r;
   880→   880→  }
   881→   881→  return fallbackResponse(identity);
   882→   882→}
