   183→   183→/** Returns "MD" for "Marie Dubois" etc. — used to derive initials from a real user name. */
   184→   184→function initialsFromName(name: string): string {
   185→   185→  const parts = name.trim().split(/\s+/).filter(Boolean);
   186→   186→  if (parts.length === 0) return "?";
   187→   187→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   188→   188→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   189→   189→}

   216→   216→/**
   217→   217→ * Adapts the backend `/api/chat` envelope (snake_case) to the frontend's
   218→   218→ * `AssistantResponse` (camelCase). Unknown / missing fields fall back to safe
   219→   219→ * defaults so the inspector never crashes.
   220→   220→ *
   221→   221→ * Field conversions:
   222→   222→ *   - `scope_clause`  → `scopeClause`
   223→   223→ *   - `tokens_in`     → `tokensIn`
   224→   224→ *   - `tokens_out`    → `tokensOut`
   225→   225→ *   - `latency_ms`    → `latencyMs`
   226→   226→ *   - `tool_calls`    → `toolCalls`
   227→   227→ *   - `security_checks` → `securityChecks` (fields are already camelCase)
   228→   228→ *   - `steps[].duration_ms` → `steps[].durationMs`
   229→   229→ */
   230→   230→function adaptBackendResponse(raw: unknown): AssistantResponse {
   231→   231→  const r =
   232→   232→    (typeof raw === "object" && raw !== null ? raw : {}) as Record<
   233→   233→      string,
   234→   234→      unknown
   235→   235→    >;
   236→   236→  const rawSteps = Array.isArray(r.steps) ? (r.steps as Record<string, unknown>[]) : [];
   237→   237→  const steps: TraceStep[] = rawSteps.map((s) => ({
   238→   238→    index: typeof s.index === "number" ? s.index : 0,
   239→   239→    title: typeof s.title === "string" ? s.title : "",
   240→   240→    detail: typeof s.detail === "string" ? s.detail : "",
   241→   241→    status: (s.status === "ok" || s.status === "warning" || s.status === "blocked"
   242→   242→      ? s.status
   243→   243→      : "ok") as TraceStep["status"],
   244→   244→    durationMs: typeof s.duration_ms === "number" ? s.duration_ms : 0,
   245→   245→  }));
   246→   246→  const rawChecks = Array.isArray(r.security_checks)
   247→   247→    ? (r.security_checks as Record<string, unknown>[])
   248→   248→    : [];
   249→   249→  const securityChecks: AssistantResponse["securityChecks"] = rawChecks.map((c) => ({
   250→   250→    label: typeof c.label === "string" ? c.label : "",
   251→   251→    status: (c.status === "ok" || c.status === "warning" || c.status === "blocked"
   252→   252→      ? c.status
   253→   253→      : "ok") as AssistantResponse["securityChecks"][number]["status"],
   254→   254→    detail: typeof c.detail === "string" ? c.detail : undefined,
   255→   255→  }));
   256→   256→  const toolCalls = Array.isArray(r.tool_calls)
   257→   257→    ? (r.tool_calls as string[])
   258→   258→    : [];
   259→   259→  return {
   260→   260→    answer: typeof r.answer === "string" ? r.answer : "Réponse vide.",
   261→   261→    sql: typeof r.sql === "string" ? r.sql : null,
   262→   262→    scopeClause:
   263→   263→      typeof r.scope_clause === "string" ? r.scope_clause : null,
   264→   264→    chart:
   265→   265→      (r.chart as AssistantResponse["chart"] | null | undefined) ?? undefined,
   266→   266→    tokensIn: typeof r.tokens_in === "number" ? r.tokens_in : 0,
   267→   267→    tokensOut: typeof r.tokens_out === "number" ? r.tokens_out : 0,
   268→   268→    latencyMs: typeof r.latency_ms === "number" ? r.latency_ms : 0,
   269→   269→    toolCalls,
   270→   270→    steps,
   271→   271→    securityChecks,
   272→   272→    refused: Boolean(r.refused),
   273→   273→  };
   274→   274→}

   276→   276→/** Calls `POST /api/chat` with the message + JWT, returns an `AssistantResponse`.
   277→   277→ *
   278→   278→ * Demo override: when `demoIdentityOverride` is set (Marie/Pierre/Admin switcher
   279→   279→ * on the demo tenant), sends body identity instead of JWT so the user can switch
   280→   280→ * without re-login. The backend's dual-mode fallback handles the scoping.
   281→   281→ */
   282→   282→async function callBackendChat(message: string): Promise<AssistantResponse> {
   283→   283→  const token = getAuthToken();
   284→   284→  const override = useCopilotStore.getState().demoIdentityOverride;
   285→   285→  const headers: Record<string, string> = {
   286→   286→    "content-type": "application/json",
   287→   287→  };
   288→   288→  // When a demo identity override is active, send body identity (no JWT) so
   289→   289→  // the backend uses the override's producer_id/role instead of the JWT's.
   290→   290→  const useJwt = token && !override;
   291→   291→  if (useJwt) headers["authorization"] = `Bearer ${token}`;
   292→   292→
   293→   293→  const body: Record<string, unknown> = { message };
   294→   294→  if (override) {
   295→   295→    body.identity_id = override.id;
   296→   296→    body.producer_id = override.producerId;
   297→   297→    body.role = override.kind;
   298→   298→  }
   299→   299→
   300→   300→  const res = await fetch("/api/chat", {
   301→   301→    method: "POST",
   302→   302→    headers,
   303→   303→    body: JSON.stringify(body),
   304→   304→  });
   305→   305→  const text = await res.text();
   306→   306→  let parsed: unknown = null;
   307→   307→  if (text.length > 0) {
   308→   308→    try {
   309→   309→      parsed = JSON.parse(text);
   310→   310→    } catch {
   311→   311→      parsed = text;
   312→   312→    }
   313→   313→  }
   314→   314→  if (!res.ok) {
   315→   315→    const message =
   316→   316→      typeof parsed === "object" && parsed && "detail" in parsed
   317→   317→        ? String((parsed as { detail: unknown }).detail)
   318→   318→        : typeof parsed === "object" && parsed && "message" in parsed
   319→   319→          ? String((parsed as { message: unknown }).message)
   320→   320→          : `Chat backend ${res.status} ${res.statusText}`;
   321→   321→    throw new Error(message);
   322→   322→  }
   323→   323→  return adaptBackendResponse(parsed);
   324→   324→}

   814→   814→function pickActiveTenant(tenants: TenantMembership[]): TenantMembership | null {
   815→   815→  if (tenants.length === 0) return null;
   816→   816→  const cachedId = getActiveTenantId();
   817→   817→  if (cachedId) {

   821→   821→  const active = tenants.find((t) => t.is_active);

   823→   823→  return tenants[0];
