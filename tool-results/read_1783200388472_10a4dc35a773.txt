   159→# Phase 5 — ML training table: daily stock snapshots per (product, date).

   161→# One row per (product_id, date) over the 90+ day history window. Each row

   170→# days_since_last_stockout, avg_sales_30d) plus the current ``stocks``
   171→# snapshot, so the live features match the training distribution.
   172→#

   223→# Documentary RAG tables (Task 23 / Phase 3 — RAG layer)
   224→# ─────────────────────────────────────────────────────────────────────────────
   225→# Two physical tables plus one FTS5 virtual table.
   226→#
   227→# ``documents`` — one row per uploaded document (PDF, text, manual note).
   228→#   ``producer_id`` is NULL for tenant-wide documents (CGV, FAQ, etc.) or
   229→#   an int for producer-private documents.
   230→#
   231→# ``document_chunks`` — paragraph-sized chunks (200-400 chars) ready for
   232→#   full-text search. Each chunk carries the same ``tenant_id`` and
   233→#   ``producer_id`` as its parent document so the RAG tool can apply
   234→#   row-level scoping at search time without joining back to ``documents``.
   235→#
   236→# ``document_chunks_fts`` — SQLite FTS5 virtual table (BM25 ranking) that
   237→#   mirrors ``document_chunks``. We chose MANUAL sync (not triggers) so
   238→#   the ingest pipeline keeps full control of the FTS content and so the
   239→#   DELETE endpoint can remove the FTS rows explicitly (FTS5 external
   240→#   content tables are trickier to keep in sync than a manual mirror).
   241→#
   242→#   We chose FTS5 over pgvector / OpenAI embeddings on purpose:
   243→#   - No external API key needed (vs OpenAI embeddings).
   244→#   - No Postgres/Docker to run (in-process SQLite only).
   245→#   - Sufficient for a small corpus (< 100 chunks per tenant).
   246→#   - Embeddings-ready architecture: ``RagSearchTool.search()`` is the
   247→#     single point of swap — replace the FTS5 SELECT with a vector
   248→#     similarity query in Phase 6 without touching the orchestrator.

   250→    "documents",
   251→    metadata,
   252→    Column("id", Integer, primary_key=True),
   253→    Column("tenant_id", String, nullable=False, default="dp", index=True),
   254→    Column("title", String, nullable=False),
   255→    Column("source_type", String, nullable=False),  # "pdf" | "text" | "manual"
   256→    Column("source_filename", String, nullable=True),
   257→    Column("content_raw", Text, nullable=False),
   258→    Column("producer_id", Integer, nullable=True, index=True),  # NULL = tenant-wide
   259→    Column("created_at", DateTime, nullable=False, default=datetime.utcnow),

   795→def _derive_product_meta(category: str, weight: int, max_weight: int) -> dict[str, Any]:
   796→    """Derive the three ML feature columns for a product row.
   797→
   798→    Returns ``{base_popularity, is_perishable, seasonality_peak_month}``.
   799→    ``base_popularity`` is normalised to 0.1..1.0 (small floor so even the
   800→    least-popular product has non-zero demand); ``is_perishable`` is True
   801→    for fast-rotating categories; ``seasonality_peak_month`` comes from
   802→    the lookup table above (None if the category is unknown).
   803→    """
   804→    norm = (weight / max_weight) if max_weight > 0 else 1.0
   805→    base_popularity = max(0.1, round(norm, 3))
   806→    return {
   807→        "base_popularity": base_popularity,
   808→        "is_perishable": category in _PERISHABLE_CATEGORIES,
   809→        "seasonality_peak_month": _CATEGORY_PEAK_MONTH.get(category),
   810→    }
   811→
   812→
   813→def _seasonality_factor(month: int, peak_month: int | None) -> float:
   814→    """Cosine seasonality factor in [0.6, 1.0].
   815→
   816→    1.0 at the peak month, 0.6 at the opposite month. Used to multiply the
   817→    daily demand so summer vegetables get a demand bump in July, etc. The
   818→    amplitude (0.2) is intentionally modest so off-season producers (e.g.
   819→    Fromagerie peaking in December) still rupture during the April-July
   820→    seed window — otherwise the RandomForest would never see a positive
   821→    example for those producers and the forecast demo would be flat.
   822→    """
   823→    if peak_month is None:
   824→        return 1.0
   825→    return 0.8 + 0.2 * math.cos((month - peak_month) * 2.0 * math.pi / 12.0)

   827→
   828→def _generate_orders_and_history(
   829→    producer_id: int,
   830→    shop_id: int,
   831→    product_rows: list[dict[str, Any]],
   832→    weights: list[int],
   833→    base_orders_per_day: float,
   834→) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
   835→    """Build orders + stock_history rows for one producer over the 91-day
   836→    history window plus July-current (2024-04-15 → 2024-07-15).
   837→
   838→    Returns ``(orders, stock_history_rows)``.
   839→
   840→    Demand model
   841→    ------------
   842→    For each day d and each product p:
   843→        daily_demand[p, d] = max(0, round(
   844→            base_popularity[p] * scale
   845→            * weekend_boost(d)
   846→            * seasonality_factor(month(d), peak_month[p])
   847→            * gauss(1, 0.2)
   848→        ))
   849→    where ``scale = 1.5`` maps base_popularity (0.1..1.0) to ~0.15..1.5
   850→    units/day, and ``weekend_boost = 1.4`` on Fri/Sat else 1.0.
   851→
   852→    Orders per day
   853→    --------------
   854→    Each day gets ``round(base_orders_per_day * gauss(1, 0.15))`` orders.
   855→    Each order picks 1-5 products weighted by popularity, with quantity 1-6.
   856→
   857→    Stock simulation
   858→    ----------------
   859→    Each product starts with ``reorder_point + random`` units. Every 7 days
   860→    (per-product restock offset) it gets ``reorder_qty`` units added. Daily
   861→    sales decrement the stock; if daily_sales > stock_level the product
   862→    ruptures that day (stockout_flag=1, stock_level → 0). Perishables get a
   863→    tighter buffer (3 days) than shelf-stable products (5 days) so they
   864→    rupture more often.
   865→    """
   866→    orders_out: list[dict[str, Any]] = []
   867→    stock_history_out: list[dict[str, Any]] = []
   868→    order_id_pool = _RNG.randint(1000, 99999)
   869→    statuses = ["pending", "paid", "ready_for_pickup", "completed", "completed", "completed", "cancelled"]
   870→
   871→    max_w = max(weights) if weights else 1
   872→    # Per-product rolling stock state.
   873→    product_state: dict[int, dict[str, Any]] = {}
   874→    for idx, p in enumerate(product_rows):
   875→        meta = _derive_product_meta(p["category"], weights[idx], max_w)
   876→        # Inject the meta into the product row so the caller can persist it.
   877→        p["base_popularity"] = meta["base_popularity"]
   878→        p["is_perishable"] = meta["is_perishable"]
   879→        p["seasonality_peak_month"] = meta["seasonality_peak_month"]
   880→        # Demand scale 2.0 (Phase 5 bump from 1.5) so even off-season producers
   881→        # see enough demand to rupture during the 91-day window.
   882→        avg_daily_demand = max(0.5, meta["base_popularity"] * 2.0)
   883→        buffer_days = 3 if meta["is_perishable"] else 4
   884→        # Restock 5 days of demand every 7 days → 2-day deficit per cycle.
   885→        # Both perishables and non-perishables use the same restock quantity
   886→        # so all producers rupture occasionally (was 5/7 split, which left
   887→        # dairy/wine/honey producers with 0 ruptures in training data).
   888→        reorder_qty = avg_daily_demand * 5
   889→        product_state[p["id"]] = {
   890→            "stock_level": avg_daily_demand * buffer_days + _RNG.uniform(0, 5),
   891→            "reorder_qty": reorder_qty,
   892→            "reorder_point": avg_daily_demand * buffer_days,
   893→            "avg_daily_demand": avg_daily_demand,
   894→            "last_stockout_day": -10,
   895→            "restock_day_offset": _RNG.randint(0, 6),
   896→        }
   897→
   898→    # Generate per-day demand for the full history window.
   899→    daily_demand: dict[tuple[int, int], float] = {}
   900→    for day_idx in range(_HISTORY_DAYS):
   901→        date = _HISTORY_START + timedelta(days=day_idx)
   902→        dow = date.weekday()
   903→        is_weekend = dow in (4, 5)
   904→        weekend_boost = 1.4 if is_weekend else 1.0
   905→        month = date.month
   906→        for p in product_rows:
   907→            # Demand scale 2.0 (Phase 5 bump from 1.5).
   908→            base = p["base_popularity"] * 2.0
   909→            seasonality = _seasonality_factor(month, p["seasonality_peak_month"])
   910→            noise = _RNG.gauss(1, 0.2)
   911→            demand = base * weekend_boost * seasonality * noise
   912→            demand = max(0, round(demand))
   913→            daily_demand[(p["id"], day_idx)] = float(demand)
   914→
   915→    # ── Generate orders per day ──
   916→    for day_idx in range(_HISTORY_DAYS):
   917→        date = _HISTORY_START + timedelta(days=day_idx)
   918→        n_orders_today = max(0, int(round(base_orders_per_day * _RNG.gauss(1, 0.15))))
   919→        for _ in range(n_orders_today):
   920→            order_id_pool += 1
   921→            ts = date + timedelta(
   922→                hours=_RNG.uniform(8, 19),
   923→                minutes=_RNG.uniform(0, 59),
   924→                seconds=_RNG.uniform(0, 59),
   925→            )
   926→            n_items = _RNG.choices([1, 2, 3, 4, 5], weights=[40, 30, 15, 10, 5])[0]
   927→            chosen_products = _RNG.choices(product_rows, weights=weights, k=n_items)
   928→            line_total = 0.0
   929→            items: list[dict[str, Any]] = []
   930→            for prod in chosen_products:
   931→                qty = float(_RNG.randint(1, 6))
   932→                price = round(prod["price_eur"] * _RNG.uniform(0.95, 1.05), 2)
   933→                lt = round(qty * price, 2)
   934→                line_total += lt
   935→                items.append(
   936→                    {
   937→                        "order_id": order_id_pool,
   938→                        "producer_id": producer_id,
   939→                        "product_id": prod["id"],
   940→                        "quantity": qty,
   941→                        "unit_price_eur": price,
   942→                        "line_total_eur": lt,
   943→                    }
   944→                )
   945→            status = _RNG.choice(statuses)
   946→            total = round(line_total, 2)
   947→            orders_out.append(
   948→                {
   949→                    "id": order_id_pool,
   950→                    "producer_id": producer_id,
   951→                    "customer_id": _RNG.randint(1, 5000),
   952→                    "shop_id": shop_id,
   953→                    "pickup_slot": ts + timedelta(hours=_RNG.randint(1, 72)),
   954→                    "total_amount": total,
   955→                    "status": status,
   956→                    "created_at": ts,
   957→                    "updated_at": ts + timedelta(hours=1),
   958→                    "items": items,
   959→                }
   960→            )
   961→
   962→        # ── Stock simulation for this day ──
   963→        dow = date.weekday()
   964→        is_weekend = dow in (4, 5)
   965→        for p in product_rows:
   966→            pid = p["id"]
   967→            state = product_state[pid]
   968→            demand_today = daily_demand[(pid, day_idx)]
   969→            stock_start = state["stock_level"]
   970→            # Restock at the start of the day if it's the restock day.
   971→            restocked = 0.0
   972→            if day_idx % 7 == state["restock_day_offset"]:
   973→                restocked = state["reorder_qty"]
   974→                stock_start += restocked
   975→            # Apply demand; rupture if demand exceeds available stock.
   976→            stockout = False
   977→            available_after = stock_start - demand_today
   978→            if available_after < 0:
   979→                stockout = True
   980→                available_after = 0.0
   981→            stock_history_out.append(
   982→                {
   983→                    "producer_id": producer_id,
   984→                    "product_id": pid,
   985→                    "date": date,
   986→                    "stock_level": round(stock_start, 2),
   987→                    "reserved": 0.0,
   988→                    "available": round(max(stock_start, 0.0), 2),
   989→                    "daily_sales": round(demand_today, 2),
   990→                    "restocked_qty": round(restocked, 2),
   991→                    "stockout_flag": stockout,
   992→                    "day_of_week": dow,
   993→                    "is_weekend": is_weekend,
   994→                    "month": date.month,
   995→                    "created_at": _REFERENCE_NOW,
   996→                }
   997→            )
   998→            state["stock_level"] = available_after
   999→            if stockout:
  1000→                state["last_stockout_day"] = day_idx
  1001→
  1002→    return orders_out, stock_history_out

  1609→def _chunk_text(content: str, target_size: int = 320, max_size: int = 480) -> list[str]:
  1610→    """Split ``content`` into paragraph-sized chunks for FTS5 indexing.
  1611→
  1612→    Strategy:
  1613→      1. Split on double newlines (paragraph boundaries).
  1614→      2. Each paragraph becomes a candidate chunk.
  1615→      3. If a paragraph is longer than ``max_size``, hard-split it at the
  1616→         last whitespace before ``target_size`` (preserves word boundaries).
  1617→      4. Discard empty chunks.
  1618→
  1619→    The 320-char target keeps chunks small enough to cite cleanly in an
  1620→    LLM answer (Phase 6) yet big enough to carry a full clause of the CGV.
  1621→    """
  1622→    chunks: list[str] = []
  1623→    for para in content.split("\n\n"):
  1624→        para = para.strip()
  1625→        if not para:
  1626→            continue
  1627→        if len(para) <= max_size:
  1628→            chunks.append(para)
  1629→            continue
  1630→        # Hard-split long paragraphs at word boundaries.
  1631→        cursor = 0
  1632→        while cursor < len(para):
  1633→            end = cursor + target_size
  1634→            if end >= len(para):
  1635→                chunks.append(para[cursor:].strip())
  1636→                break
  1637→            # Walk back to the previous whitespace.
  1638→            while end > cursor and para[end] not in (" ", "\n"):
  1639→                end -= 1
  1640→            if end == cursor:  # single word longer than target_size
  1641→                end = cursor + target_size
  1642→            chunks.append(para[cursor:end].strip())
  1643→            cursor = end
  1644→    return [c for c in chunks if c]

  1647→async def ingest_document(
  1648→    tenant_id: str,
  1649→    title: str,
  1650→    source_type: str,
  1651→    content: str,
  1652→    producer_id: int | None = None,
  1653→    source_filename: str | None = None,
  1654→) -> int:
  1655→    """Insert a document + its chunks, sync the FTS5 table, return doc id.
  1656→
  1657→    Called by:
  1658→      - ``init_db()`` to seed the 4 fictitious DP documents.
  1659→      - ``POST /api/documents`` to ingest an uploaded PDF or text file.
  1660→
  1661→    The FTS5 table is synced manually (not via triggers) — see the comment
  1662→    on ``document_chunks_fts`` above for the rationale.
  1663→    """
  1664→    engine = get_engine()
  1665→    now = datetime.utcnow()
  1666→    chunks = _chunk_text(content)
  1667→    async with engine.begin() as conn:
  1668→        result = await conn.execute(
  1669→            documents.insert().values(
  1670→                tenant_id=tenant_id,
  1671→                title=title,
  1672→                source_type=source_type,
  1673→                source_filename=source_filename,
  1674→                content_raw=content,
  1675→                producer_id=producer_id,
  1676→                created_at=now,
  1677→            ).returning(documents.c.id)
  1678→        )
  1679→        doc_id = result.scalar_one()
  1680→        for idx, chunk in enumerate(chunks):
  1681→            await conn.execute(
  1682→                document_chunks.insert().values(
  1683→                    document_id=doc_id,
  1684→                    chunk_index=idx,
  1685→                    content=chunk,
  1686→                    tenant_id=tenant_id,
  1687→                    producer_id=producer_id,
  1688→                    created_at=now,
  1689→                )
  1690→            )
  1691→            # Manual FTS5 sync.
  1692→            await conn.execute(
  1693→                text(
  1694→                    "INSERT INTO document_chunks_fts "
  1695→                    "(content, document_id, tenant_id, producer_id) "
  1696→                    "VALUES (:content, :doc_id, :tenant_id, :producer_id)"
  1697→                ),
  1698→                {
  1699→                    "content": chunk,
  1700→                    "doc_id": doc_id,
  1701→                    "tenant_id": tenant_id,
  1702→                    "producer_id": producer_id,
  1703→                },
  1704→            )
  1705→    logger.info(
  1706→        "ingest_document — doc_id=%d title=%r chunks=%d tenant=%s producer=%s",
  1707→        doc_id, title, len(chunks), tenant_id, producer_id,
  1708→    )
  1709→    return doc_id

  1712→async def delete_document(doc_id: int) -> bool:
  1713→    """Delete a document, its chunks, and its FTS5 rows. Returns True if deleted.
  1714→
  1715→    Order matters for FTS5: delete from the virtual table first (it has no
  1716→    FK constraints), then ``document_chunks``, then ``documents``.
  1717→    """
  1718→    engine = get_engine()
  1719→    async with engine.begin() as conn:
  1720→        # Verify existence first (so we can return False cleanly).
  1721→        existing = await conn.execute(
  1722→            select(documents.c.id).where(documents.c.id == doc_id)
  1723→        )
  1724→        if existing.fetchone() is None:
  1725→            return False
  1726→        # 1. FTS5 virtual table (manual sync — no cascade).
  1727→        await conn.execute(
  1728→            text("DELETE FROM document_chunks_fts WHERE document_id = :doc_id"),
  1729→            {"doc_id": doc_id},
  1730→        )
  1731→        # 2. document_chunks.
  1732→        await conn.execute(
  1733→            document_chunks.delete().where(document_chunks.c.document_id == doc_id)
  1734→        )
  1735→        # 3. documents.
  1736→        await conn.execute(
  1737→            documents.delete().where(documents.c.id == doc_id)
  1738→        )
  1739→    logger.info("delete_document — doc_id=%d deleted", doc_id)
  1740→    return True
