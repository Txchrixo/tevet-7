   180→/** Returns "MD" for "Marie Dubois" etc. — used to derive initials from a real user name. */
   181→function initialsFromName(name: string): string {
   182→  const parts = name.trim().split(/\s+/).filter(Boolean);
   183→  if (parts.length === 0) return "?";
   184→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   185→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   186→}

   213→/**
   214→ * Adapts the backend `/api/chat` envelope (snake_case) to the frontend's
   215→ * `AssistantResponse` (camelCase). Unknown / missing fields fall back to safe
   216→ * defaults so the inspector never crashes.
   217→ *
   218→ * Field conversions:
   219→ *   - `scope_clause`  → `scopeClause`
   220→ *   - `tokens_in`     → `tokensIn`
   221→ *   - `tokens_out`    → `tokensOut`
   222→ *   - `latency_ms`    → `latencyMs`
   223→ *   - `tool_calls`    → `toolCalls`
   224→ *   - `security_checks` → `securityChecks` (fields are already camelCase)
   225→ *   - `steps[].duration_ms` → `steps[].durationMs`
   226→ */
   227→function adaptBackendResponse(raw: unknown): AssistantResponse {
   228→  const r =
   229→    (typeof raw === "object" && raw !== null ? raw : {}) as Record<
   230→      string,
   231→      unknown
   232→    >;
   233→  const rawSteps = Array.isArray(r.steps) ? (r.steps as Record<string, unknown>[]) : [];
   234→  const steps: TraceStep[] = rawSteps.map((s) => ({
   235→    index: typeof s.index === "number" ? s.index : 0,
   236→    title: typeof s.title === "string" ? s.title : "",
   237→    detail: typeof s.detail === "string" ? s.detail : "",
   238→    status: (s.status === "ok" || s.status === "warning" || s.status === "blocked"
   239→      ? s.status
   240→      : "ok") as TraceStep["status"],
   241→    durationMs: typeof s.duration_ms === "number" ? s.duration_ms : 0,
   242→  }));
   243→  const rawChecks = Array.isArray(r.security_checks)
   244→    ? (r.security_checks as Record<string, unknown>[])
   245→    : [];
   246→  const securityChecks: AssistantResponse["securityChecks"] = rawChecks.map((c) => ({
   247→    label: typeof c.label === "string" ? c.label : "",
   248→    status: (c.status === "ok" || c.status === "warning" || c.status === "blocked"
   249→      ? c.status
   250→      : "ok") as AssistantResponse["securityChecks"][number]["status"],
   251→    detail: typeof c.detail === "string" ? c.detail : undefined,
   252→  }));
   253→  const toolCalls = Array.isArray(r.tool_calls)
   254→    ? (r.tool_calls as string[])
   255→    : [];
   256→  return {
   257→    answer: typeof r.answer === "string" ? r.answer : "Réponse vide.",
   258→    sql: typeof r.sql === "string" ? r.sql : null,
   259→    scopeClause:
   260→      typeof r.scope_clause === "string" ? r.scope_clause : null,
   261→    chart:
   262→      (r.chart as AssistantResponse["chart"] | null | undefined) ?? undefined,
   263→    tokensIn: typeof r.tokens_in === "number" ? r.tokens_in : 0,
   264→    tokensOut: typeof r.tokens_out === "number" ? r.tokens_out : 0,
   265→    latencyMs: typeof r.latency_ms === "number" ? r.latency_ms : 0,
   266→    toolCalls,
   267→    steps,
   268→    securityChecks,
   269→    refused: Boolean(r.refused),
   270→  };
   271→}

   273→/** Calls `POST /api/chat` with the message + JWT, returns an `AssistantResponse`.
   274→ *
   275→ * Demo override: when `demoIdentityOverride` is set (Marie/Pierre/Admin switcher
   276→ * on the demo tenant), sends body identity instead of JWT so the user can switch
   277→ * without re-login. The backend's dual-mode fallback handles the scoping.
   278→ */
   279→async function callBackendChat(message: string): Promise<AssistantResponse> {
   280→  const token = getAuthToken();
   281→  const override = useCopilotStore.getState().demoIdentityOverride;
   282→  const headers: Record<string, string> = {
   283→    "content-type": "application/json",
   284→  };
   285→  // When a demo identity override is active, send body identity (no JWT) so
   286→  // the backend uses the override's producer_id/role instead of the JWT's.
   287→  const useJwt = token && !override;
   288→  if (useJwt) headers["authorization"] = `Bearer ${token}`;
   289→
   290→  const body: Record<string, unknown> = { message };
   291→  if (override) {
   292→    body.identity_id = override.id;
   293→    body.producer_id = override.producerId;
   294→    body.role = override.kind;
   295→  }
   296→
   297→  const res = await fetch("/api/chat", {
   298→    method: "POST",
   299→    headers,
   300→    body: JSON.stringify(body),
   301→  });
   302→  const text = await res.text();
   303→  let parsed: unknown = null;
   304→  if (text.length > 0) {
   305→    try {
   306→      parsed = JSON.parse(text);
   307→    } catch {
   308→      parsed = text;
   309→    }
   310→  }
   311→  if (!res.ok) {
   312→    const message =
   313→      typeof parsed === "object" && parsed && "detail" in parsed
   314→        ? String((parsed as { detail: unknown }).detail)
   315→        : typeof parsed === "object" && parsed && "message" in parsed
   316→          ? String((parsed as { message: unknown }).message)
   317→          : `Chat backend ${res.status} ${res.statusText}`;
   318→    throw new Error(message);
   319→  }
   320→  return adaptBackendResponse(parsed);
   321→}

   347→  /**
   348→   * Called on app start. If we have a stored JWT, validate it via `/api/auth/me`
   349→   * and load the user's tenants. If the JWT is invalid or absent, switch to
   350→   * `anonymous` so the AuthScreen renders. Network failures fall back to
   351→   * `anonymous` (the user can still click "Essayer la démo").
   352→   */
   353→  bootstrap: async () => {
   354→    const token = getAuthToken();
   355→    if (!token) {
   356→      set({ authMode: "anonymous" });
   357→      return;
   358→    }
   359→    try {
   360→      const me = await getMe();
   361→      const tenants = me.memberships ?? [];
   362→      const activeTenant = pickActiveTenant(tenants);
   363→      set({
   364→        authMode: "authenticated",
   365→        isDemoSession: false,
   366→        user: me.user,
   367→        tenants,
   368→        activeTenant,
   369→        identity: identityFromAuthUser(me.user, activeTenant),
   370→      });
   371→    } catch (err) {
   372→      // Stale or invalid JWT — clear it and show the auth screen.
   373→      setAuthToken(null);
   374→      setActiveTenantId(null);
   375→      set({ authMode: "anonymous", user: null, tenants: [], activeTenant: null });
   376→      if (err instanceof AuthApiError && !err.unreachable) {
   377→        // 401 / 403 etc. — silently clear. Other errors (502) are also silent
   378→        // so the user just sees the auth screen.
   379→      }
   380→    }
   381→  },
   382→
   383→  /**
   384→   * Real backend login. Stores the JWT, loads `/api/auth/me` + `/api/tenants/mine`,
   385→   * activates the first tenant if none is active, then switches to `authenticated`.
   386→   *
   387→   * Returns `{ ok: false, error }` on failure so the AuthScreen can show an
   388→   * inline error message. Network failures (backend unreachable) are surfaced
   389→   * as `error` so the caller can decide whether to fall back to demo mode.
   390→   */

   397→      const me = await getMe();

   399→      const activeTenant = pickActiveTenant(tenants);
   400→      if (activeTenant) setActiveTenantId(activeTenant.tenant_id);
   401→      set({
   402→        authMode: "authenticated",
   403→        isDemoSession: false,
   404→        authLoading: false,
   405→        user,
   406→        tenants,
   407→        activeTenant,
   408→        identity: identityFromAuthUser(user, activeTenant),

   550→   * Switches the active tenant. Calls `POST /api/tenants/{id}/activate` to get

   552→   */
   553→  switchTenant: async (tenantId) => {
   554→    const token = getAuthToken();
   555→    if (!token) return;
   556→    try {
   557→      const { membership, token: newToken } = await activateTenant(tenantId);
   558→      setAuthToken(newToken);
   559→      setActiveTenantId(membership.tenant_id);
   560→      const tenants = get().tenants.map((t) =>
   561→        t.tenant_id === membership.tenant_id
   562→          ? { ...t, is_active: true }
   563→          : { ...t, is_active: false },
   564→      );
   565→      const activeTenant = tenants.find((t) => t.tenant_id === membership.tenant_id) ?? null;
   566→      const user = get().user;
   567→      set({
   568→        tenants,
   569→        activeTenant,
   570→        identity: user ? identityFromAuthUser(user, activeTenant) : get().identity,
   571→        messages: [],
   572→        selectedMessageId: null,
   573→        isStreaming: false,
   574→        inspectorOpen: false,
   575→        adminView: "none",
   576→      });
   577→      toast.success(`Tenant activé : ${activeTenant?.name ?? tenantId}`);
   578→    } catch (err) {
   579→      const message =
   580→        err instanceof AuthApiError
   581→          ? err.unreachable
   582→            ? "Backend injoignable"
   583→            : err.message
   584→          : err instanceof Error
   585→            ? err.message
   586→            : "Erreur inconnue";
   587→      toast.error("Impossible de changer de tenant", { description: message });
   588→    }
   589→  },

   599→      set({

   603→        messages: [],
   604→        selectedMessageId: null,
   605→        isStreaming: false,
   606→        inspectorOpen: false,
   607→        adminView: "none",
   608→      });
   609→      toast.success(`Workspace créé : ${name}`);

   803→/**
   804→ * Picks the tenant to activate after login: the cached `activeTenantId` if the
   805→ * user is still a member, else the first `is_active` membership, else the
   806→ * first membership. Returns null if the user has no memberships.
   807→ */
   808→function pickActiveTenant(tenants: TenantMembership[]): TenantMembership | null {
   809→  if (tenants.length === 0) return null;
   810→  const cachedId = getActiveTenantId();
   811→  if (cachedId) {
   812→    const cached = tenants.find((t) => t.tenant_id === cachedId);
   813→    if (cached) return cached;
   814→  }
   815→  const active = tenants.find((t) => t.is_active);
   816→  if (active) return active;
   817→  return tenants[0];
   818→}

   820→/**
   821→ * Shared helper that pushes the user message, simulates the "thinking" delay,
   822→ * then reveals the assistant response. Responses are computed up-front from
   823→ * the mock layer so they stay deterministic per identity + question.
   824→ */
   825→function runAssistant(
   826→  get: () => CopilotState,
   827→  set: (partial: Partial<CopilotState>) => void,
   828→  userText: string,
   829→  computeResponse: () => AssistantResponse,
   830→) {
   831→  if (get().isStreaming) return;
   832→
   833→  const userMessage: ChatMessage = {
   834→    id: makeId("u"),
   835→    role: "user",
   836→    content: userText,
   837→    createdAt: Date.now(),
   838→  };
   839→
   840→  set({
   841→    messages: [...get().messages, userMessage],
   842→    isStreaming: true,
   843→    selectedMessageId: null,
   844→  });
   845→
   846→  // Simulate model latency before revealing the answer.
   847→  const delay = 800 + Math.random() * 350;
   848→  window.setTimeout(() => {
   849→    const response = computeResponse();
   850→    const assistantMessage: ChatMessage = {
   851→      id: makeId("a"),
   852→      role: "assistant",
   853→      content: response.answer,
   854→      response,
   855→      createdAt: Date.now(),
   856→      streaming: false,
   857→    };
   858→    // If the inspector is already open, live-update it with the new trace.
   859→    // Otherwise leave it closed — the user can click the answer to inspect.
   860→    const inspectorWasOpen = get().inspectorOpen;
   861→    set({
   862→      messages: [...get().messages, assistantMessage],
   863→      isStreaming: false,
   864→      selectedMessageId: inspectorWasOpen ? assistantMessage.id : null,
   865→    });
   866→  }, delay);
   867→}

   869→/**
   870→ * Backend-backed assistant run. Pushes the user message immediately, shows the
   871→ * typing indicator, calls `/api/chat` with the JWT, then pushes the assistant
   872→ * message with the real envelope (SQL, chart, steps, security_checks).
   873→ *
   874→ * Network / HTTP errors are surfaced as a synthetic assistant message so the
   875→ * chat thread doesn't break.
   876→ */
   877→async function runBackendAssistant(
   878→  get: () => CopilotState,
   879→  set: (partial: Partial<CopilotState>) => void,
   880→  userText: string,
   881→) {
   882→  if (get().isStreaming) return;
   883→
   884→  const userMessage: ChatMessage = {
   885→    id: makeId("u"),
   886→    role: "user",
   887→    content: userText,
   888→    createdAt: Date.now(),
   889→  };
   890→
   891→  set({
   892→    messages: [...get().messages, userMessage],
   893→    isStreaming: true,
   894→    selectedMessageId: null,
   895→  });
   896→
   897→  try {
   898→    const response = await callBackendChat(userText);
   899→    const assistantMessage: ChatMessage = {
   900→      id: makeId("a"),
   901→      role: "assistant",
   902→      content: response.answer,
   903→      response,
   904→      createdAt: Date.now(),
   905→      streaming: false,
   906→    };
   907→    const inspectorWasOpen = get().inspectorOpen;
   908→    set({
   909→      messages: [...get().messages, assistantMessage],
   910→      isStreaming: false,
   911→      selectedMessageId: inspectorWasOpen ? assistantMessage.id : null,
   912→    });
   913→  } catch (err) {
   914→    const message =
   915→      err instanceof Error ? err.message : "Erreur inconnue";
   916→    const failureResponse: AssistantResponse = {
   917→      answer:
   918→        "⚠️ Le service agent n'a pas pu répondre.\n\n" +
   919→        `Détail : ${message}\n\n` +
   920→        "Réessayez dans un instant. Si le problème persiste, basculez en mode démo via le menu utilisateur.",
   921→      sql: null,
   922→      scopeClause: null,
   923→      tokensIn: 0,
   924→      tokensOut: 0,
   925→      latencyMs: 0,
   926→      toolCalls: [],
   927→      steps: [],
   928→      securityChecks: [],
   929→      refused: true,
   930→    };
   931→    const assistantMessage: ChatMessage = {
   932→      id: makeId("a"),
   933→      role: "assistant",
   934→      content: failureResponse.answer,
   935→      response: failureResponse,
   936→      createdAt: Date.now(),
   937→      streaming: false,
   938→    };
   939→    set({
   940→      messages: [...get().messages, assistantMessage],
   941→      isStreaming: false,
   942→    });
   943→    toast.error("Échec de l'agent", { description: message });
   944→  }
   945→}
